```python
def bil(a_f, b_f, free, ndof, edof, Ke0):
    """Element bilinear form a_e^T Ke0 b_e (free-DOF vectors embedded to full)."""
    a = np.zeros(ndof); a[free] = a_f
    b = np.zeros(ndof); b[free] = b_f
    return np.einsum("ei,ij,ej->e", a[edof], Ke0, b[edof])

def adjoint_grad(s, K, M, C, U, V, A, w, dE, dm, aR, bR, free, ndof, edof, K0, M0):
    am, af = s.alpha_m, s.alpha_f
    a0, a1, a2, a3, a4, a5, a6, a7 = s.a0, s.a1, s.a2, s.a3, s.a4, s.a5, s.a6, s.a7
    pm, qm, rm = (1-am)*a0, (1-am)*a2, (1-am)*a3 - am
    pc, qc, rc = (1-af)*a1, (1-af)*a4 - af, (1-af)*a5
    kK = (1-af) + a1*(1-af)*bR          # dK_hat/drho_e = dE_e*kK*K0 + dm_e*kM*M0
    kM = a0*(1-am) + a1*(1-af)*aR
    N = U.shape[0] - 1
    grad = np.zeros(len(dE))
    for n in range(N + 1):               # explicit term dJ/drho at fixed u
        if w[n]:
            grad += w[n] * dE * bil(U[n], U[n], free, ndof, edof, K0)
    lu = np.zeros(U.shape[1]); lv = np.zeros_like(lu); la = np.zeros_like(lu)
    for n in range(N - 1, -1, -1):       # step n -> n+1, reversed
        lu = lu + 2.0 * w[n+1] * (K @ U[n+1])
        la_e = la + a7 * lv              # v_{n+1} depends on a_{n+1}
        lu_e = lu + a0 * la_e            # a_{n+1} depends on u_{n+1}
        mu = s.pardiso_solver.solve(s.K_hat_ff, lu_e)   # K_hat symmetric
        rm_v = pm*U[n] + qm*V[n] + rm*A[n]
        rc_v = pc*U[n] + qc*V[n] + rc*A[n]
        grad += dE * bil(mu, -af*U[n] + bR*rc_v - kK*U[n+1], free, ndof, edof, K0)
        grad += dm * bil(mu, rm_v + aR*rc_v - kM*U[n+1], free, ndof, edof, M0)
        lu_n = -af*(K @ mu) + (pm*M + pc*C) @ mu - a0*la_e
        lv_n = (qm*M + qc*C) @ mu - a2*la_e + lv
        la = (rm*M + rc*C) @ mu - a3*la_e + a6*lv
        lu, lv = lu_n, lv_n
    return grad   # dJ/drho_phys; dE = p*rho^(p-1)*(E0-Emin), dm = rho_mat (linear)
```

```python
def distribute_wrench(r, F, Mp):
    """r: (n,3) node offsets from ref point P [m]; F [N], Mp [N*m] about P."""
    rbar = r.mean(0); rc = r - rbar
    Mg = Mp - np.cross(rbar, F)                     # moment about centroid
    J = np.einsum("ni,ni->", rc, rc) * np.eye(3) - rc.T @ rc
    mu = np.linalg.solve(J, Mg)
    f = F / len(r) + np.cross(mu, rc)               # min-norm, exact sum F and moment Mp
    assert np.allclose(f.sum(0), F) and np.allclose(np.cross(r, f).sum(0), Mp)
    return f
```
